cmake_minimum_required(VERSION 3.25)
project(ProjectWhiteVdbSpike LANGUAGES CXX)
find_package(OpenVDB CONFIG REQUIRED)
add_executable(white_vdb_spike main.cpp)
target_compile_features(white_vdb_spike PRIVATE cxx_std_20)
target_link_libraries(white_vdb_spike PRIVATE OpenVDB::openvdb)
include(CTest)
if(BUILD_TESTING)
  add_test(NAME vdb_roundtrip COMMAND white_vdb_spike --self-test "${CMAKE_CURRENT_BINARY_DIR}/evidence")
endif()
