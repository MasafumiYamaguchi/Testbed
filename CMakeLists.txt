cmake_minimum_required(VERSION 3.25)
project(ProjectWhite VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC AND MSVC_VERSION LESS 1930)
  message(FATAL_ERROR "ProjectWhite requires Visual Studio 2022 (MSVC 19.30+) and a Windows 10/11 SDK")
elseif(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 11)
  message(FATAL_ERROR "ProjectWhite CPU tests require GCC 11+ with C++20")
endif()

option(WHITE_BUILD_APP "Build the SDL desktop application" ON)
option(WHITE_BUILD_GPU_SPIKE "Use the Windows SDL GPU validation application" ${WIN32})
option(WHITE_BUILD_VDB_SPIKE "Build the independent OpenVDB spike (Issue #13)" OFF)
if(WHITE_BUILD_VDB_SPIKE)
  add_subdirectory(spikes/vdb)
endif()

include(FetchContent)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_Declare(nlohmann_json
  GIT_REPOSITORY https://github.com/nlohmann/json.git
  GIT_TAG 55f93686c01528224f448c19128836e7df245f72)
FetchContent_MakeAvailable(nlohmann_json)
FetchContent_Declare(tinyexr GIT_REPOSITORY https://github.com/syoyo/tinyexr.git
  GIT_TAG 735ff73ce5959cf005eb99ce517c9bcecab89dfb SOURCE_SUBDIR white-unused)
FetchContent_MakeAvailable(tinyexr)
add_library(white_exr STATIC src/core/exr_codec.cpp ${tinyexr_SOURCE_DIR}/deps/miniz/miniz.c)
target_include_directories(white_exr PRIVATE ${tinyexr_SOURCE_DIR} ${tinyexr_SOURCE_DIR}/deps/miniz PUBLIC include)
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
  OUTPUT_VARIABLE WHITE_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE)
if(NOT WHITE_COMMIT)
  set(WHITE_COMMIT "unknown")
endif()
add_library(white_core src/core/build_info.cpp src/core/document.cpp src/core/persistence.cpp src/core/density.cpp src/core/optics.cpp src/core/editor_geometry.cpp src/core/noise.cpp src/core/dense_cache.cpp src/core/revision_queue.cpp src/core/benchmark.cpp src/core/phase.cpp src/core/hdr_export.cpp src/core/sun_cache.cpp)
target_link_libraries(white_core PRIVATE nlohmann_json::nlohmann_json white_exr)
target_compile_definitions(white_core PRIVATE WHITE_COMMIT="${WHITE_COMMIT}")
target_include_directories(white_core PUBLIC include)
target_compile_features(white_core PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(white_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(white_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(WHITE_BUILD_APP)
  include(cmake/Dependencies.cmake)
  if(WHITE_BUILD_GPU_SPIKE)
    if(NOT WIN32)
      message(FATAL_ERROR "The GPU spike currently validates Windows/D3D12 only. Set WHITE_BUILD_GPU_SPIKE=OFF for the bootstrap window.")
    endif()
    FetchContent_MakeAvailable(imgui)
    add_library(white_imgui STATIC
      ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp
      ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp
      ${imgui_SOURCE_DIR}/backends/imgui_impl_sdl3.cpp
      ${imgui_SOURCE_DIR}/backends/imgui_impl_sdlgpu3.cpp)
    target_include_directories(white_imgui PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
    target_link_libraries(white_imgui PUBLIC SDL3::SDL3)
    FetchContent_MakeAvailable(imguizmo)
    add_library(white_gizmo STATIC ${imguizmo_SOURCE_DIR}/src/ImGuizmo.cpp)
    target_include_directories(white_gizmo PUBLIC ${imguizmo_SOURCE_DIR}/src)
    target_link_libraries(white_gizmo PUBLIC white_imgui)
    add_executable(white_app src/app/gpu_main.cpp src/app/editor_ui.cpp src/gpu/spike.cpp src/gpu/bake_worker.cpp)
    target_link_libraries(white_app PRIVATE white_core white_imgui white_gizmo)
    include(cmake/Shaders.cmake)
    white_compile_shader(field.comp.hlsl cs_6_0)
    white_compile_shader(sample.comp.hlsl cs_6_0)
    white_compile_shader(fullscreen.vert.hlsl vs_6_0)
    white_compile_shader(slice.frag.hlsl ps_6_0)
    white_compile_shader(volume.frag.hlsl ps_6_0)
    white_compile_shader(tonemap.frag.hlsl ps_6_0)
    white_compile_shader(optics.comp.hlsl cs_6_0)
    white_compile_shader(cache_samples.comp.hlsl cs_6_0)
    white_compile_shader(sun_cache.comp.hlsl cs_6_0)
  else()
    add_executable(white_app src/app/main.cpp)
    target_link_libraries(white_app PRIVATE white_core SDL3::SDL3)
  endif()
  if(WIN32)
    add_custom_command(TARGET white_app POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL3::SDL3> $<TARGET_FILE_DIR:white_app>)
  endif()
endif()
