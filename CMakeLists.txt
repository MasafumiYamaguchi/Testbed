cmake_minimum_required(VERSION 3.25)
project(ProjectWhite VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC AND MSVC_VERSION LESS 1930)
  message(FATAL_ERROR "ProjectWhite requires Visual Studio 2022 (MSVC 19.30+) and a Windows 10/11 SDK")
elseif(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 11)
  message(FATAL_ERROR "ProjectWhite CPU tests require GCC 11+ with C++20")
endif()

option(WHITE_BUILD_APP "Build the SDL desktop application" ON)
option(WHITE_BUILD_VDB_SPIKE "Build the independent OpenVDB spike (Issue #13)" OFF)
if(WHITE_BUILD_VDB_SPIKE)
  message(FATAL_ERROR "OpenVDB spike is not implemented yet; see Issue #13. Leave WHITE_BUILD_VDB_SPIKE=OFF for the desktop app.")
endif()

add_library(white_core src/core/build_info.cpp)
target_include_directories(white_core PUBLIC include)
target_compile_features(white_core PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(white_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(white_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(WHITE_BUILD_APP)
  include(cmake/Dependencies.cmake)
  add_executable(white_app src/app/main.cpp)
  target_link_libraries(white_app PRIVATE white_core SDL3::SDL3)
  if(WIN32)
    add_custom_command(TARGET white_app POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL3::SDL3> $<TARGET_FILE_DIR:white_app>)
  endif()
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(white_core_tests tests/core_tests.cpp)
  target_link_libraries(white_core_tests PRIVATE white_core)
  add_test(NAME core_contract COMMAND white_core_tests)
  if(WHITE_BUILD_APP)
    add_test(NAME app_help COMMAND white_app --help)
    add_test(NAME app_invalid_option COMMAND white_app --unknown-option)
    set_tests_properties(app_invalid_option PROPERTIES WILL_FAIL TRUE)
  endif()
endif()
